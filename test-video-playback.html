Test 2: Console Debug Script

Run this in the browser console after uploading a video:

// Debug video playback
const video = document.querySelector('video');
if (video) {
    console.log('Video found:', {
        src: video.src,
        duration: video.duration,
        paused: video.paused,
        muted: video.muted,
        controls: video.controls,
        readyState: video.readyState,
        networkState: video.networkState,
        error: video.error
    });

    // Try to play
    video.play().then(() => {
        console.log('✅ Video playing successfully');
    }).catch(err => {
        console.error('❌ Play failed:', err);
        console.log('Try clicking on the video or unmuting first');
    });
} else {
    console.log('❌ No video element found');
}

// Check for play buttons
const playButtons = document.querySelectorAll('[data-testid="play-button"]');
console.log('Play buttons found:', playButtons.length);
playButtons.forEach((btn, i) => {
    console.log(`Button ${i}:`, btn);
    // Simulate click
    btn.click();
});

Test 3: Check Z-Index Issues

// Check for overlapping elements
const elements = document.querySelectorAll('*');
const videoRect = document.querySelector('video')?.getBoundingClientRect();

if (videoRect) {
    const overlapping = [];
    elements.forEach(el => {
        const rect = el.getBoundingClientRect();
        if (rect.left < videoRect.right &&
            rect.right > videoRect.left &&
            rect.top < videoRect.bottom &&
            rect.bottom > videoRect.top &&
            el !== document.querySelector('video')) {
            const zIndex = window.getComputedStyle(el).zIndex;
            const pointerEvents = window.getComputedStyle(el).pointerEvents;
            overlapping.push({
                element: el,
                class: el.className,
                zIndex,
                pointerEvents
            });
        }
    });
    console.table(overlapping);
}

Test 4: Fix Playback Issues

// Force fix video playback
const fixPlayback = () => {
    const video = document.querySelector('video');
    if (!video) {
        console.log('No video found');
        return;
    }

    // Remove muted attribute temporarily
    const wasMuted = video.muted;
    video.muted = false;

    // Add user interaction handler
    const playVideo = async () => {
        try {
            // Try with sound first
            await video.play();
            console.log('✅ Playing with sound');
        } catch (err) {
            // Fallback to muted
            video.muted = true;
            try {
                await video.play();
                console.log('✅ Playing muted');
            } catch (err2) {
                console.error('❌ Cannot play:', err2);
            }
        }
    };

    // Try immediate play
    playVideo();

    // Add click handler to video
    video.addEventListener('click', playVideo);

    // Fix pointer events on overlays
    document.querySelectorAll('.absolute.inset-0').forEach(el => {
        if (!el.querySelector('button')) {
            el.style.pointerEvents = 'none';
        }
    });

    console.log('🔧 Playback fixes applied');
};

fixPlayback();
